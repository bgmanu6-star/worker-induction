        // --- NEW: CSV / Excel Bulk Importer Function ---
        function handleCSVImport(event) {
            const file = event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            showToast("Reading file and importing to cloud...");

            reader.onload = async (e) => {
                try {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, { type: 'array' });
                    const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                    const rows = XLSX.utils.sheet_to_json(firstSheet);

        function createTinyThumbnail(file, maxWidth = 60, maxHeight = 60) {
            return new Promise((resolve) => {
                if (!file) return resolve("");
                const reader = new FileReader();
                reader.onload = (e) => {
                    const img = new Image();
                    img.onload = () => {
                        const canvas = document.createElement('canvas');
                        let w = img.width, h = img.height;
                        if (w > h) { if (w > maxWidth) { h *= maxWidth / w; w = maxWidth; } }
                        else { if (h > maxHeight) { w *= maxHeight / h; h = maxHeight; } }
                        canvas.width = w; canvas.height = h;
                        const ctx = canvas.getContext('2d');
                        ctx.drawImage(img, 0, 0, w, h);
                        resolve(canvas.toDataURL('image/jpeg', 0.6));
                    };
                    img.src = e.target.result;
                };
                reader.readAsDataURL(file);
            });
        }

        async function uploadToFreeImageHost(file) {
            const apiKey = localStorage.getItem('imgbb_api_key');
            if (!apiKey || !file) return "";
            try {
                const formData = new FormData();
                formData.append("image", file);
                const res = await fetch(`https://api.imgbb.com/1/upload?key=${apiKey}`, { method: "POST", body: formData });
                const data = await res.json();
                if (data && data.success) return data.data.url;
            } catch (err) { console.error(err); }
            return "";
        }

        async function uploadToTelegram(file, caption) {
            const botToken = localStorage.getItem('tg_bot_token');
            const chatId = localStorage.getItem('tg_chat_id');
            if (!botToken || !chatId || !file) return "";
            try {
                const formData = new FormData();
                formData.append("chat_id", chatId);
                formData.append("photo", file);
                formData.append("caption", caption);
                const res = await fetch(`https://api.telegram.org/bot${botToken}/sendPhoto`, { method: "POST", body: formData });
                const data = await res.json();
                if (data && data.ok) {
                    const cleanChatId = String(chatId).replace("-100", "");
                    return `https://t.me/c/${cleanChatId}/${data.result.message_id}`;
                }
            } catch (err) { console.error(err); }
            return "";
        }

        async function handleProofUpload(event, side) {
            const file = event.target.files[0];
            if (!file) return;
            if (side === 'front') {
                selectedFiles.proofFront = file;
                const prev = document.getElementById('proofFrontPreview');
                prev.src = URL.createObjectURL(file); prev.classList.remove('hidden');
            } else {
                selectedFiles.proofBack = file;
                const prev = document.getElementById('proofBackPreview');
                prev.src = URL.createObjectURL(file); prev.classList.remove('hidden');
            }
            const ocrStatus = document.getElementById('ocrStatus');
            ocrStatus.classList.remove('hidden');
            showToast(`Scanning ID ${side}...`);
            try {
                const result = await Tesseract.recognize(file, 'eng');
                parseOcrTextToForm(result.data.text, side);
                showToast("ID scanned & fields auto-filled!");
            } catch (err) {
                showToast("Could not auto-read text.", "error");
            } finally {
                ocrStatus.classList.add('hidden');
            }
        }

        function parseOcrTextToForm(rawText, side) {
            const lines = rawText.split('\n').map(l => l.trim()).filter(l => l.length > 2);
            const idMatch = rawText.match(/\b\d{4}\s?\d{4}\s?\d{4}\b/);
            if (idMatch && !document.getElementById('aadhaar').value) {
                document.getElementById('aadhaar').value = idMatch[0];
            }
            const dobMatch = rawText.match(/\b(\d{2})[\/\-](\d{2})[\/\-](\d{4})\b/);
            if (dobMatch && !document.getElementById('dob').value) {
                document.getElementById('dob').value = `${dobMatch[3]}-${dobMatch[2]}-${dobMatch[1]}`;
                calculateAge();
            }
            const pinMatch = rawText.match(/\b[654321]\d{5}\b/);
            if (pinMatch && !document.getElementById('permPincode').value) {
                const pin = pinMatch[0];
                document.getElementById('permPincode').value = pin;
                handlePincodeInput(pin, 'permanentAddress', 'permLastFetched');
            }
            const fatherRegex = /(?:S\/O|C\/O|W\/O|D\/O|Father)\s*[:\-]?\s*([A-Za-z\s\.]+)/i;
            const fatherMatch = rawText.match(fatherRegex);
            if (fatherMatch && fatherMatch[1] && !document.getElementById('fatherName').value) {
                document.getElementById('fatherName').value = fatherMatch[1].split('\n')[0].trim();
            }
        }

        function generateQRCodeBase64(text) {
            try {
                const qr = qrcode(0, 'M');
                qr.addData(text);
                qr.make();
                const cellSize = 3, margin = 6;
                const size = qr.getModuleCount() * cellSize + margin * 2;
                const qrCanvas = document.createElement('canvas');
                qrCanvas.width = size; qrCanvas.height = size;
                const qCtx = qrCanvas.getContext('2d');
                qCtx.fillStyle = '#ffffff'; qCtx.fillRect(0, 0, size, size);
                qCtx.fillStyle = '#000000';
                for (let r = 0; r < qr.getModuleCount(); r++) {
                    for (let c = 0; c < qr.getModuleCount(); c++) {
                        if (qr.isDark(r, c)) qCtx.fillRect(margin + c * cellSize, margin + r * cellSize, cellSize, cellSize);
                    }
                }
                return qrCanvas.toDataURL('image/png');
            } catch (err) { return ""; }
        }

        async function handlePincodeInput(pincode, targetFieldId, lastFetchedId) {
            const inputField = document.getElementById(targetFieldId);
            const lastFetchedInput = document.getElementById(lastFetchedId);
            let currentVal = inputField.value.trim();
            if (pincode.length !== 6) return;
            try {
                const response = await fetch(`https://api.postalpincode.in/pincode/${pincode}`);
                const data = await response.json();
                if (data[0].Status === "Success") {

        }

        function calculateAge() {
            const dobVal = document.getElementById('dob').value;
            if (!dobVal) { document.getElementById('age').value = ''; return; }
            const dobDate = new Date(dobVal);
            if (isNaN(dobDate)) return;
            const today = new Date();
            let age = today.getFullYear() - dobDate.getFullYear();
            const m = today.getMonth() - dobDate.getMonth();
            if (m < 0 || (m === 0 && today.getDate() < dobDate.getDate())) age--;
            document.getElementById('age').value = age >= 0 ? age + ' Years' : '0 Years';
        }

        function isAadhaarDuplicated(aadhaarStr) {

            return workers.filter(w => (w.aadhaar || '').trim() === clean).length > 1;
        }
